2.) Write a C program for implementation of Round Robin scheduling algorithms.

Ans:-

AIM
To write a C program to implement Round Robin scheduling algorithm.

ALGORITHM
1. Read number of processes, burst time, and time quantum.
2. Initialize remaining burst time for each process.
3. Execute each process for time quantum in cyclic order.
4. Update remaining time and calculate waiting & turnaround time.
5. Display average waiting time and turnaround time.

PROGRAM

#include <stdio.h>

int main() {
    int n, i, time = 0, remain, tq;
    int wt = 0, tat = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    int bt[n], rt[n];

    for(i = 0; i < n; i++) {
        printf("Enter burst time for process %d: ", i+1);
        scanf("%d", &bt[i]);
        rt[i] = bt[i]; // remaining time
    }

    printf("Enter time quantum: ");
    scanf("%d", &tq);

    remain = n;

    while(remain != 0) {
        for(i = 0; i < n; i++) {
            if(rt[i] > 0) {
                if(rt[i] <= tq) {
                    time += rt[i];
                    wt += time - bt[i];
                    tat += time;
                    rt[i] = 0;
                    remain--;
                } else {
                    time += tq;
                    rt[i] -= tq;
                }
            }
        }
    }

    printf("Average Waiting Time = %.2f\n", (float)wt/n);
    printf("Average Turnaround Time = %.2f\n", (float)tat/n);

    return 0;
}

AIM
To write a C program to implement First Come First Serve (FCFS) scheduling algorithm.

ALGORITHM
1. Read number of processes and their burst times.
2. Assume arrival order is the same as input order.
3. Calculate waiting time for each process.
4. Calculate turnaround time for each process.
5. Compute and display average waiting time and turnaround time.

PROGRAM

#include <stdio.h>

int main() {
    int n, i;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    int bt[n], wt[n], tat[n];
    float total_wt = 0, total_tat = 0;

    // Input burst times
    for(i = 0; i < n; i++) {
        printf("Enter burst time for process %d: ", i+1);

    // First process waiting time is 0
    wt[0] = 0;

    // Calculate waiting time
    for(i = 1; i < n; i++) {
        wt[i] = wt[i-1] + bt[i-1];

    // Calculate turnaround time
    for(i = 0; i < n; i++) {
        tat[i] = wt[i] + bt[i];

    // Calculate totals
    for(i = 0; i < n; i++) {

    int n, f, i, j, k, flag, faults = 0;

    printf("Enter number of pages: ");

    printf("Enter page reference string: ");

        scanf("%d", &pages[i]);

    printf("Enter number of frames: ");

    int frames[f];

    // Initialize frames to -1 (empty)
    for(i = 0; i < f; i++) {
        frames[i] = -1;

    int index = 0;

    for(i = 0; i < n; i++) {

                flag = 1;

                printf("%d ", frames[k]);

4. Update shared variable and unlock the mutex.

PROGRAM

#include <stdio.h>
#include <pthread.h>

int counter = 0;           // shared resource
pthread_mutex_t lock;      // mutex

void* increment(void* arg) {
    for(int i = 0; i < 100000; i++) {
        pthread_mutex_lock(&lock);    // lock
        counter++;                    // critical section
        pthread_mutex_unlock(&lock);  // unlock
    }
    return NULL;

    pthread_t t1, t2;

    pthread_mutex_init(&lock, NULL);

    // Create threads
    pthread_create(&t1, NULL, increment, NULL);
    pthread_create(&t2, NULL, increment, NULL);

    // Wait for threads to finish
    pthread_join(t1, NULL);
    pthread_join(t2, NULL);

    printf("Final Counter Value = %d\n", counter);

    pthread_mutex_destroy(&lock);

    return 0;
}

PROGRAM

#include <stdio.h>

int main() {
    int nb, np, i, j;

    printf("Enter number of memory blocks: ");

    int block[nb];

8.) Write a C program to implement Banker's algorithm for deadlock avoidance.

Ans:-

AIM
To write a C program to implement Banker's algorithm for deadlock avoidance.

ALGORITHM
1. Read number of processes, resources, Allocation, Max, and Available matrices.
2. Compute Need = Max - Allocation.
3. Initialize Finish array to false and Work = Available.
4. Find a process whose Need <= Work, mark it finished and update Work.
5. Repeat until all processes finish (safe state) or no such process exists (unsafe).

PROGRAM

#include <stdio.h>

int main() {
    int n, m, i, j, k;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    printf("Enter number of resources: ");
    scanf("%d", &m);

    int alloc[n][m], max[n][m], need[n][m];
    int avail[m];

    printf("Enter Allocation Matrix:\n");
    for(i = 0; i < n; i++)
        for(j = 0; j < m; j++)
            scanf("%d", &alloc[i][j]);

    printf("Enter Max Matrix:\n");
    for(i = 0; i < n; i++)
        for(j = 0; j < m; j++)
            scanf("%d", &max[i][j]);

    printf("Enter Available Resources:\n");
    for(i = 0; i < m; i++)
        scanf("%d", &avail[i]);

    // Calculate Need matrix
    for(i = 0; i < n; i++)
        for(j = 0; j < m; j++)
            need[i][j] = max[i][j] - alloc[i][j];

    int finish[n], safeSeq[n];
    for(i = 0; i < n; i++)
        finish[i] = 0;

    int work[m];
    for(i = 0; i < m; i++)
        work[i] = avail[i];

    int count = 0;

    while(count < n) {
        int found = 0;

        for(i = 0; i < n; i++) {
            if(finish[i] == 0) {
                for(j = 0; j < m; j++) {
                    if(need[i][j] > work[j])
                        break;
                }

                if(j == m) {
                    for(k = 0; k < m; k++)
                        work[k] += alloc[i][k];

                    safeSeq[count++] = i;
                    finish[i] = 1;
                    found = 1;
                }
            }
        }

        if(found == 0) {
            printf("System is not in safe state\n");
            return 0;
        }
    }

    printf("System is in safe state\nSafe sequence: ");
    for(i = 0; i < n; i++)
        printf("P%d ", safeSeq[i]);

    printf("\n");

    return 0;
}

10.) Write a C program to implement the producer-consumer problem using semaphores.

Ans:-

AIM
To write a C program to implement Producer-Consumer problem using semaphores.

ALGORITHM
1. Initialize buffer, semaphores (empty, full) and mutex.
2. Create producer and consumer threads.
3. Producer waits on empty, locks mutex, adds item, signals full.
4. Consumer waits on full, locks mutex, removes item, signals empty.
5. Repeat for fixed number of items and display output.

PROGRAM

#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>

#define SIZE 5

int buffer[SIZE];
int in = 0, out = 0;

sem_t empty, full;
pthread_mutex_t mutex;

void* producer(void* arg) {
    int item;
    for(int i = 1; i <= 10; i++) {
        item = i;

        sem_wait(&empty);
        pthread_mutex_lock(&mutex);

        buffer[in] = item;
        printf("Produced: %d\n", item);
        in = (in + 1) % SIZE;

        pthread_mutex_unlock(&mutex);
        sem_post(&full);
    }
    return NULL;
}

void* consumer(void* arg) {
    int item;
    for(int i = 1; i <= 10; i++) {

        sem_wait(&full);
        pthread_mutex_lock(&mutex);

        item = buffer[out];
        printf("Consumed: %d\n", item);
        out = (out + 1) % SIZE;

        pthread_mutex_unlock(&mutex);
        sem_post(&empty);
    }
    return NULL;
}

int main() {
    pthread_t p, c;

    sem_init(&empty, 0, SIZE);
    sem_init(&full, 0, 0);
    pthread_mutex_init(&mutex, NULL);

    pthread_create(&p, NULL, producer, NULL);
    pthread_create(&c, NULL, consumer, NULL);

    pthread_join(p, NULL);
    pthread_join(c, NULL);

    sem_destroy(&empty);
    sem_destroy(&full);
    pthread_mutex_destroy(&mutex);

    return 0;
}

11.) Write a C program for implementation of SJF scheduling algorithm.

Ans:-

AIM
To write a C program to implement Shortest Job First (SJF) scheduling algorithm.

ALGORITHM
1. Read number of processes and their burst times.
2. Sort processes in ascending order of burst time.
3. Calculate waiting time for each process.
4. Calculate turnaround time for each process.
5. Compute and display average waiting and turnaround time.

PROGRAM

#include <stdio.h>

int main() {
    int n, i, j;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    int bt[n], wt[n], tat[n], p[n];

    // Input burst times
    for(i = 0; i < n; i++) {
        printf("Enter burst time for process %d: ", i+1);
        scanf("%d", &bt[i]);
        p[i] = i + 1;
    }

    // Sort processes by burst time (SJF)
    for(i = 0; i < n; i++) {
        for(j = i + 1; j < n; j++) {
            if(bt[i] > bt[j]) {
                // Swap burst time
                int temp = bt[i];
                bt[i] = bt[j];
                bt[j] = temp;

                // Swap process numbers
                temp = p[i];
                p[i] = p[j];
                p[j] = temp;
            }
        }
    }

    // Calculate waiting time
    wt[0] = 0;
    for(i = 1; i < n; i++) {
        wt[i] = wt[i-1] + bt[i-1];
    }

    // Calculate turnaround time
    for(i = 0; i < n; i++) {
        tat[i] = wt[i] + bt[i];
    }

    float total_wt = 0, total_tat = 0;

    printf("\nProcess\tBT\tWT\tTAT\n");
    for(i = 0; i < n; i++) {
        printf("P%d\t%d\t%d\t%d\n", p[i], bt[i], wt[i], tat[i]);
        total_wt += wt[i];
        total_tat += tat[i];
    }

    printf("\nAverage Waiting Time = %.2f", total_wt / n);
    printf("\nAverage Turnaround Time = %.2f\n", total_tat / n);

    return 0;
}

        read(fd[0], read_msg, sizeof(read_msg));

        close(fd[0]);
    }

    return 0;

14.) Write a C program to implement LRU page replacement algorithm.

Ans:-

AIM
To write a C program to implement Least Recently Used (LRU) page replacement algorithm.

ALGORITHM
1. Read number of pages, page reference string, and number of frames.
2. Initialize frames as empty and track recent usage.
3. For each page, check if it is already in frames.
4. If not present, replace the least recently used page.
5. Count and display total page faults.

PROGRAM

#include <stdio.h>

int main() {
    int n, f, i, j, k, pos, min, faults = 0;

    printf("Enter number of pages: ");
    scanf("%d", &n);

    int pages[n];
    printf("Enter page reference string:\n");
    for(i = 0; i < n; i++) {
        scanf("%d", &pages[i]);
    }

    printf("Enter number of frames: ");
    scanf("%d", &f);

    int frames[f], recent[f];

    // Initialize frames
    for(i = 0; i < f; i++) {
        frames[i] = -1;
        recent[i] = -1;
    }

    printf("\nPage\tFrames\n");

    for(i = 0; i < n; i++) {
        int found = 0;

        // Check if page already exists
        for(j = 0; j < f; j++) {
            if(frames[j] == pages[i]) {
                found = 1;
                recent[j] = i; // update recent use
                break;
            }
        }

        // If not found -> page fault
        if(!found) {
            faults++;

            // Find empty frame
            for(j = 0; j < f; j++) {
                if(frames[j] == -1) {
                    pos = j;
                    break;
                }
            }

            // If no empty frame, find LRU
            if(j == f) {
                min = recent[0];
                pos = 0;

                for(k = 1; k < f; k++) {
                    if(recent[k] < min) {
                        min = recent[k];
                        pos = k;
                    }
                }
            }

            frames[pos] = pages[i];
            recent[pos] = i;
        }

        // Display frames
        printf("%d\t", pages[i]);
        for(j = 0; j < f; j++) {
            if(frames[j] != -1)
                printf("%d ", frames[j]);
            else
                printf("- ");
        }
        printf("\n");
    }

    printf("\nTotal Page Faults = %d\n", faults);

    return 0;
}

ALGORITHM

#include <stdio.h>
#include <stdlib.h>

int main() {
    int n, i, j, min, pos, head, seek = 0;

    printf("Enter number of requests: ");
    scanf("%d", &n);

    int req[n], visited[n];

    printf("Enter request sequence:\n");
    for(i = 0; i < n; i++) {
        scanf("%d", &req[i]);
        visited[i] = 0;
    }

17.) Write a C program to implement LFU page replacement algorithm.

Ans:-

AIM
To write a C program to implement Least Frequently Used (LFU) page replacement algorithm.

ALGORITHM
1. Read number of pages, page reference string, and number of frames.
2. Initialize frames, frequency array, and page fault counter.
3. For each page, check if it is already in frames.
4. If not present, replace the page with the least frequency.
5. If tie occurs, replace the least recently used among them.

PROGRAM

#include <stdio.h>

int main() {
    int n, f, i, j, min, pos, faults = 0;

    printf("Enter number of pages: ");
    scanf("%d", &n);

    int pages[n];
    printf("Enter page reference string:\n");
    for(i = 0; i < n; i++) {
        scanf("%d", &pages[i]);
    }

    printf("Enter number of frames: ");
    scanf("%d", &f);

    int frames[f], freq[f], time[f];

    // Initialize
    for(i = 0; i < f; i++) {
        frames[i] = -1;
        freq[i] = 0;
        time[i] = 0;
    }

    printf("\nPage\tFrames\n");

    for(i = 0; i < n; i++) {
        int found = 0;

        // Check if page already exists
        for(j = 0; j < f; j++) {
            if(frames[j] == pages[i]) {
                freq[j]++;
                found = 1;
                time[j] = i;
                break;
            }
        }

        // Page fault
        if(!found) {
            faults++;

            // Find empty frame
            for(j = 0; j < f; j++) {
                if(frames[j] == -1) {
                    pos = j;
                    break;
                }
            }

            // If no empty frame, apply LFU
            if(j == f) {
                min = freq[0];
                pos = 0;

                for(j = 1; j < f; j++) {
                    if(freq[j] < min) {
                        min = freq[j];
                        pos = j;
                    }
                    // Tie breaker: LRU
                    else if(freq[j] == min && time[j] < time[pos]) {
                        pos = j;
                    }
                }
            }

            frames[pos] = pages[i];
            freq[pos] = 1;
            time[pos] = i;
        }

        // Display frames
        printf("%d\t", pages[i]);
        for(j = 0; j < f; j++) {
            if(frames[j] != -1)
                printf("%d ", frames[j]);
            else
                printf("- ");
        }
        printf("\n");
    }

    printf("\nTotal Page Faults = %d\n", faults);

    return 0;
}

19.) Write a C program to implement SCAN disk scheduling algorithm.

Ans:-

AIM
To write a C program to implement SCAN (Elevator) disk scheduling algorithm.

ALGORITHM
1. Read number of requests, request sequence, initial head position, and disk size.
2. Sort the request array in ascending order.
3. Move head towards one end (0 or max disk) servicing requests.
4. After reaching the end, reverse direction and continue servicing.
5. Calculate and display total seek time.

PROGRAM

#include <stdio.h>
#include <stdlib.h>

int main() {
    int n, i, j, head, disk_size, seek = 0, direction;

    printf("Enter number of requests: ");
    scanf("%d", &n);

    int req[n + 2];

    printf("Enter request sequence:\n");
    for(i = 0; i < n; i++) {
        scanf("%d", &req[i]);
    }

    printf("Enter initial head position: ");
    scanf("%d", &head);

    printf("Enter disk size: ");
    scanf("%d", &disk_size);

    printf("Enter direction (0 = left, 1 = right): ");
    scanf("%d", &direction);

    // Add boundary values
    req[n] = head;
    req[n + 1] = (direction == 1) ? disk_size - 1 : 0;

    n += 2;

    // Sort requests
    for(i = 0; i < n; i++) {
        for(j = i + 1; j < n; j++) {
            if(req[i] > req[j]) {
                int temp = req[i];
                req[i] = req[j];
                req[j] = temp;
            }
        }
    }

    // Find head position
    int pos;
    for(i = 0; i < n; i++) {
        if(req[i] == head) {
            pos = i;
            break;
        }
    }

    printf("\nSeek Sequence:\n");

    if(direction == 1) {
        // Move right
        for(i = pos; i < n; i++) {
            printf("%d ", req[i]);
        }
        for(i = pos - 1; i >= 0; i--) {
            printf("%d ", req[i]);
        }
    } else {
        // Move left
        for(i = pos; i >= 0; i--) {
            printf("%d ", req[i]);
        }
        for(i = pos + 1; i < n; i++) {
            printf("%d ", req[i]);
        }
    }

    // Calculate seek time
    seek = abs(head - req[0]) + abs(req[n - 1] - req[0]);

    printf("\nTotal Seek Time = %d\n", seek);

    return 0;
}
