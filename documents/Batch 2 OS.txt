1.) Write a C program to implement UNIX system calls and file management.

Ans:-

AIM
To write a C program to demonstrate UNIX system calls for file management
(create, write, read, and close a file).

ALGORITHM
1. Create a file using open() system call.
2. Write data into the file using write().
3. Close the file.
4. Reopen the file in read mode.
5. Read and display the file contents using read().

PROGRAM

#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
#include <string.h>

int main() {
    int fd;
    char write_msg[] = "Hello, this is UNIX file handling program\n";
    char read_msg[100];

    // Create file
    fd = open("file.txt", O_CREAT | O_WRONLY, 0644);
    if(fd == -1) {
        printf("File creation failed\n");
        return 1;
    }

    // Write into file
    write(fd, write_msg, strlen(write_msg));

    // Close file
    close(fd);

    // Reopen file in read mode
    fd = open("file.txt", O_RDONLY);
    if(fd == -1) {
        printf("File open failed\n");
        return 1;
    }

    // Read file content
    int n = read(fd, read_msg, sizeof(read_msg));

    // Null terminate string
    read_msg[n] = '\0';

    printf("File Content:\n%s", read_msg);

    close(fd);

    return 0;
}

OUTPUT
File Content:
Hello, this is UNIX file handling program

The program was successfully executed and verified.

4.) Write a C program to implement first fit, best fit algorithm for memory management.

Ans:-

AIM
To write a C program to implement First Fit and Best Fit memory allocation algorithms.

ALGORITHM
1. Read number of memory blocks and their sizes.
2. Read number of processes and their sizes.
3. Apply First Fit: allocate first suitable block.
4. Apply Best Fit: allocate smallest suitable block.
5. Display allocation results for both methods.

PROGRAM

#include <stdio.h>

void firstFit(int block[], int m, int process[], int n) {
    int alloc[n];

    for(int i = 0; i < n; i++)
        alloc[i] = -1;

    for(int i = 0; i < n; i++) {
        for(int j = 0; j < m; j++) {
            if(block[j] >= process[i]) {
                alloc[i] = j;
                block[j] -= process[i];
                break;
            }
        }
    }

    printf("\nFirst Fit Allocation:\n");
    printf("Process\tSize\tBlock\n");

    for(int i = 0; i < n; i++) {
        printf("%d\t%d\t", i+1, process[i]);
        if(alloc[i] != -1)
            printf("%d\n", alloc[i]+1);
        else
            printf("Not Allocated\n");
    }
}

void bestFit(int block[], int m, int process[], int n) {
    int alloc[n];

    for(int i = 0; i < n; i++)
        alloc[i] = -1;

    for(int i = 0; i < n; i++) {
        int bestIdx = -1;

        for(int j = 0; j < m; j++) {
            if(block[j] >= process[i]) {
                if(bestIdx == -1 || block[j] < block[bestIdx]) {
                    bestIdx = j;
                }
            }
        }

        if(bestIdx != -1) {
            alloc[i] = bestIdx;
            block[bestIdx] -= process[i];
        }
    }

    printf("\nBest Fit Allocation:\n");
    printf("Process\tSize\tBlock\n");

    for(int i = 0; i < n; i++) {
        printf("%d\t%d\t", i+1, process[i]);
        if(alloc[i] != -1)
            printf("%d\n", alloc[i]+1);
        else
            printf("Not Allocated\n");
    }
}

int main() {
    int m, n;

    printf("Enter number of blocks: ");
    scanf("%d", &m);

    int block1[m], block2[m];

    printf("Enter block sizes:\n");
    for(int i = 0; i < m; i++) {
        scanf("%d", &block1[i]);
        block2[i] = block1[i];
    }

    printf("Enter number of processes: ");
    scanf("%d", &n);

    int process[n];

    printf("Enter process sizes:\n");
    for(int i = 0; i < n; i++) {
        scanf("%d", &process[i]);
    }

    firstFit(block1, m, process, n);
    bestFit(block2, m, process, n);

    return 0;
}

To write a C program demonstrating wait(), exit(), and stat() system calls in UNIX.

ALGORITHM
1. Create a child process using fork().

5. Display process and file details.

PROGRAM

#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>
#include <sys/stat.h>
#include <stdlib.h>

int main() {
    pid_t pid;
    struct stat st;

    pid = fork();

    if(pid == 0) {
        printf("Child PID: %d\n", getpid());
        exit(0);

        wait(NULL);

        printf("Parent PID: %d\n", getpid());

        stat("file.txt", &st);
        printf("File size: %ld bytes\n", st.st_size);

    return 0;
}

13.) Write a program to simulate UNIX commands CP and ls.

Ans:-

AIM
To write a C program to simulate UNIX commands cp (copy file) and ls (list directory contents).

ALGORITHM
1. Accept choice from user (cp or ls).
2. If cp, read source and destination file names.
3. Copy contents from source file to destination file.
4. If ls, display list of files in current directory.
5. Execute using system calls and display output.

PROGRAM

#include <stdio.h>
#include <stdlib.h>
#include <dirent.h>

void copyFile() {
    FILE *fp1, *fp2;
    char ch;
    char source[30], dest[30];

    printf("Enter source file: ");
    scanf("%s", source);

    printf("Enter destination file: ");
    scanf("%s", dest);

    fp1 = fopen(source, "r");
    if(fp1 == NULL) {
        printf("Source file not found\n");
        return;
    }

    fp2 = fopen(dest, "w");

    while((ch = fgetc(fp1)) != EOF) {
        fputc(ch, fp2);
    }

    printf("File copied successfully\n");

    fclose(fp1);
    fclose(fp2);
}

void listFiles() {
    struct dirent *d;
    DIR *dir = opendir(".");

    if(dir == NULL) {
        printf("Directory not found\n");
        return;
    }

    printf("\nFiles in current directory:\n");

    while((d = readdir(dir)) != NULL) {
        printf("%s\n", d->d_name);
    }

    closedir(dir);
}

int main() {
    int choice;

    printf("1. cp (copy file)\n");
    printf("2. ls (list files)\n");
    printf("Enter choice: ");
    scanf("%d", &choice);

    switch(choice) {
        case 1:
            copyFile();
            break;

        case 2:
            listFiles();
            break;

        default:
            printf("Invalid choice\n");
    }

    return 0;
}

15.) Write a C program to implement the Round Robin (RR) CPU scheduling.

Ans:-

AIM
To write a C program to implement Round Robin CPU scheduling algorithm.

ALGORITHM
1. Read number of processes, burst times, and time quantum.
2. Initialize remaining burst time for each process.
3. Execute processes in cyclic order using time quantum.
4. Update remaining time and calculate waiting time.
5. Compute and display waiting and turnaround times.

PROGRAM

#include <stdio.h>

int main() {
    int n, i, tq, time = 0, remain;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    int bt[n], rt[n], wt[n], tat[n];

    printf("Enter burst times:\n");
    for(i = 0; i < n; i++) {
        scanf("%d", &bt[i]);
        rt[i] = bt[i];
        wt[i] = 0;
    }

    printf("Enter time quantum: ");
    scanf("%d", &tq);

    remain = n;

    while(remain > 0) {
        for(i = 0; i < n; i++) {
            if(rt[i] > 0) {
                if(rt[i] <= tq) {
                    time += rt[i];
                    rt[i] = 0;
                    remain--;
                    wt[i] = time - bt[i];
                } else {
                    time += tq;
                    rt[i] -= tq;
                }
            }
        }
    }

    float total_wt = 0, total_tat = 0;

    printf("\nProcess\tBT\tWT\tTAT\n");

    for(i = 0; i < n; i++) {
        tat[i] = bt[i] + wt[i];
        total_wt += wt[i];
        total_tat += tat[i];

        printf("P%d\t%d\t%d\t%d\n", i+1, bt[i], wt[i], tat[i]);
    }

    printf("\nAverage Waiting Time = %.2f", total_wt / n);
    printf("\nAverage Turnaround Time = %.2f\n", total_tat / n);

    return 0;
}

OUTPUT

Enter time quantum: 2

Process  BT   WT   TAT
P1       5    6    11
P2       4    6    10
P3       2    4    6

Average Waiting Time = 5.33
Average Turnaround Time = 9.00

20.) Write a C program to simulate Paging technique for memory management.

Ans:-

AIM
To write a C program to simulate paging technique for memory management.

ALGORITHM
1. Read size of logical memory, page size, and physical memory.
2. Calculate number of pages and frames.
3. Read page table mapping (page to frame).
4. Accept logical address (page number + offset).
5. Convert to physical address and display result.

PROGRAM

#include <stdio.h>

int main() {
    int logical_mem, physical_mem, page_size;
    int pages, frames;

    printf("Enter logical memory size: ");
    scanf("%d", &logical_mem);

    printf("Enter physical memory size: ");
    scanf("%d", &physical_mem);

    printf("Enter page size: ");
    scanf("%d", &page_size);

    pages = logical_mem / page_size;
    frames = physical_mem / page_size;

    int page_table[pages];

    printf("\nEnter page table (frame mapping):\n");
    for(int i = 0; i < pages; i++) {
        printf("Page %d -> Frame: ", i);
        scanf("%d", &page_table[i]);
    }

    int page_no, offset;

    printf("\nEnter logical address (page number & offset):\n");
    scanf("%d %d", &page_no, &offset);

    if(page_no >= pages || offset >= page_size) {
        printf("Invalid logical address\n");
        return 0;
    }

    int frame_no = page_table[page_no];
    int physical_address = frame_no * page_size + offset;

    printf("\nPhysical Address = %d\n", physical_address);

    return 0;
}

OUTPUT
Enter logical memory size: 16
